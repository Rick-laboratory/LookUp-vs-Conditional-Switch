cmake_minimum_required(VERSION 3.16)
project(LookupVsSwitch LANGUAGES CXX)
add_executable(dispatch-benchmark main.cpp)
target_compile_features(dispatch-benchmark PRIVATE cxx_std_17)
set_target_properties(dispatch-benchmark PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(dispatch-benchmark PRIVATE /W4 /permissive-)
else()
    target_compile_options(dispatch-benchmark PRIVATE -Wall -Wextra -Wpedantic)
endif()
